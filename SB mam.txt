module rca(a,b,cin,sum,cout);
input a,b,cin;
output sum,cout;

assign sum=a^b^cin;
assign cout=(a&b)|(b&cin)|(a&cin);

endmodule


module ripple_carry_adder(A,B,Cin,SUM,Cout);

input [7:0] A,B;
input Cin;
output [7:0] SUM;
output Cout;

wire C1,C2,C3,C4,C5,C6,C7;

rca FA0(A[0],B[0],Cin,SUM[0],C1);
rca FA1(A[1],B[1],C1,SUM[1],C2);
rca FA2(A[2],B[2],C2,SUM[2],C3);
rca FA3(A[3],B[3],C3,SUM[3],C4);
rca FA4(A[4],B[4],C4,SUM[4],C5);
rca FA5(A[5],B[5],C5,SUM[5],C6);
rca FA6(A[6],B[6],C6,SUM[6],C7);
rca FA7(A[7],B[7],C7,SUM[7],Cout);

endmodule
